2. Componente Sello QR Baulia y Estados (Lógica React): Justo arriba de tu componente App, declara este componente vectorial que dibuja el QR con puntos circulares, marcos squircle, monograma central y validador matemático de contraste (>40%):
JavaScript

const BauliaSignatureQR = ({ qrUrl, color = '#1C1917', logoSrc = 'monograma.svg' }) => {
  const [logoOk, setLogoOk] = React.useState(true);
  const { modules, count, safeColor, eyeColor } = React.useMemo(() => {
    const ensureContrast = (hex) => {
      let clean = String(hex || '#1C1917').replace('#', '').trim();
      if (clean.length === 3) clean = clean.split('').map(c => c + c).join('');
      let r = parseInt(clean.substring(0, 2), 16), g = parseInt(clean.substring(2, 4), 16), b = parseInt(clean.substring(4, 6), 16);
      if (isNaN(r) || isNaN(g) || isNaN(b)) return { main: '#1C1917', eye: '#1C1917' };
      let lum = (0.299 * r + 0.587 * g + 0.114 * b) / 255;
      while (lum > 0.50) { r = Math.floor(r * 0.82); g = Math.floor(g * 0.82); b = Math.floor(b * 0.82); lum = (0.299 * r + 0.587 * g + 0.114 * b) / 255; }
      const toHex = (n) => n.toString(16).padStart(2, '0');
      return { main: `#${toHex(r)}${toHex(g)}${toHex(b)}`, eye: `#${toHex(Math.floor(r * 0.65))}${toHex(Math.floor(g * 0.65))}${toHex(Math.floor(b * 0.65))}` };
    };
    const colors = ensureContrast(color);
    let payload = qrUrl || 'https://baulia.com';
    try { if (payload.includes('data=')) payload = decodeURIComponent(payload.split('data=')[1].split('&')[0]); } catch (e) {}
    if (typeof window.qrcode !== 'function') return { modules: [], count: 0, safeColor: colors.main, eyeColor: colors.eye };
    const qr = window.qrcode(0, 'H'); qr.addData(payload); qr.make();
    const n = qr.getModuleCount(), dots = [], center = n / 2, clearRadius = n * 0.135;
    for (let row = 0; row < n; row++) {
      for (let col = 0; col < n; col++) {
        if ((row < 7 && col < 7) || (row < 7 && col >= n - 7) || (row >= n - 7 && col < 7)) continue;
        if (logoOk && logoSrc && Math.abs(row + 0.5 - center) < clearRadius && Math.abs(col + 0.5 - center) < clearRadius) continue;
        if (qr.isDark(row, col)) dots.push({ r: row, c: col });
      }
    }
    return { modules: dots, count: n, safeColor: colors.main, eyeColor: colors.eye };
  }, [qrUrl, color, logoOk, logoSrc]);
  if (!count) return <img src={qrUrl} alt="QR" crossOrigin="anonymous" className="w-40 h-40 mix-blend-multiply block" />;
  const renderEye = (x, y) => (<g key={`${x}-${y}`}><rect x={x + 0.5} y={y + 0.5} width="6" height="6" rx="1.85" fill="none" stroke={eyeColor} strokeWidth="1" /><circle cx={x + 3.5} cy={y + 3.5} r="1.5" fill={safeColor} /></g>);
  return (
    <div className="relative w-40 h-40 flex items-center justify-center select-none">
      <svg viewBox={`0 0 ${count} ${count}`} width="160" height="160" className="w-full h-full block">{renderEye(0, 0)}{renderEye(count - 7, 0)}{renderEye(0, count - 7)}{modules.map((m, i) => (<circle key={i} cx={m.c + 0.5} cy={m.r + 0.5} r="0.42" fill={safeColor} />))}</svg>
      {logoOk && logoSrc && (<div className="absolute inset-0 flex items-center justify-center pointer-events-none"><div className="w-9 h-9 rounded-full bg-white shadow-md border border-[#D4AF37]/50 p-1 flex items-center justify-center"><img src={logoSrc} alt="Monograma" crossOrigin="anonymous" className="w-full h-full object-contain" onError={() => setLogoOk(false)} /></div></div>)}
    </div>
  );
};

const descargarPase = async (id, name) => {
  try {
    const html2canvas = window.html2canvas;
    const element = document.getElementById(`ticket_${id}`);
    const canvas = await html2canvas(element, { scale: 3, useCORS: true, backgroundColor: null });
    canvas.toBlob(async (blob) => {
      const file = new File([blob], `Pase_${name.replace(/ /g, '_')}.png`, { type: 'image/png' });
      if (navigator.canShare && navigator.canShare({ files: [file] })) {
        await navigator.share({ files: [file], title: 'Mi Pase VIP' });
      } else {
        const link = document.createElement('a'); link.href = URL.createObjectURL(blob); link.download = file.name; link.click();
      }
    }, 'image/png');
  } catch (e) { console.error("Error al descargar:", e); }
};
